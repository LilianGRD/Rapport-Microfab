% Dose D = I * t avec I = 3,11 mW/cm2 (lampe P.250)
% Epaisseur normalisee = e / e0 (e0 : tableau 1.1)
% Ajustements lineaires e/e0 = s*log10(D) + c sur la zone de transition :
%   2000 tr/min, points 1 a 4 : s = -0,8607 ; c = 1,0669 ; D100 = 1,196 ; D0 = 17,36
%   3000 tr/min, points 1 a 4 : s = -1,4894 ; c = 1,6811 ; D100 = 2,866 ; D0 = 13,45

\pgfmathsetmacro{\Ilampe}{3.11}
\begin{tikzpicture}
  \begin{axis}[
      width=0.9\linewidth, height=6.6cm,
      xmode=log, xmin=0.7, xmax=40,
      log ticks with fixed point, xtick={1,2,3,5,10,20,30},
      ymin=-0.05, ymax=1.05,
      xlabel={Dose $D = I\,t$ (\unit{\milli\joule\per\centi\metre\squared}), échelle logarithmique},
      ylabel={Épaisseur normalisée $e/e_0$},
      grid=both, grid style={gray!20}, minor grid style={gray!10},
      legend pos=north east, legend cell align=left,
      legend style={draw=none, font=\footnotesize, fill=white, fill opacity=0.85, text opacity=1},
      yticklabel style={/pgf/number format/fixed, /pgf/number format/precision=1},
      every axis plot/.append style={thin},
    ]
    % --- donnees ---
    \addplot[c2000, mark=*, mark size=1.8pt, mark options={fill=c2000}]
      table[x expr=\Ilampe*\thisrow{t23}, y expr=\thisrow{e2000}/1.8193] {data/sujet1_contraste.dat};
    \addlegendentry{\qty{2000}{\tour\per\minute} ($e_0 = \qty{1,82}{\micro\metre}$)}
    \addplot[c3000, mark=square*, mark size=1.8pt, mark options={fill=c3000}]
      table[x expr=\Ilampe*\thisrow{t23}, y expr=\thisrow{e3000}/1.4242] {data/sujet1_contraste.dat};
    \addlegendentry{\qty{3000}{\tour\per\minute} ($e_0 = \qty{1,42}{\micro\metre}$)}
    \addplot[c4000, mark=triangle*, mark size=2.2pt, mark options={fill=c4000}]
      table[x expr=\Ilampe*\thisrow{t45}, y expr=\thisrow{e4000}/1.2113] {data/sujet1_contraste.dat};
    \addlegendentry{\qty{4000}{\tour\per\minute} ($e_0 = \qty{1,21}{\micro\metre}$)}
    \addplot[c5000, mark=diamond*, mark size=2.2pt, mark options={fill=c5000}]
      table[x expr=\Ilampe*\thisrow{t45}, y expr=\thisrow{e5000}/1.2956] {data/sujet1_contraste.dat};
    \addlegendentry{\qty{5000}{\tour\per\minute} ($e_0 = \qty{1,30}{\micro\metre}$)}

    % --- ajustements de la zone de transition ---
    \addplot[c2000, dashed, thick, domain=1.1958:17.3605, samples=2, forget plot] {-0.8607*log10(x)+1.0669};
    \addplot[c3000, dashed, thick, domain=2.8664:13.4513, samples=2, forget plot] {-1.4894*log10(x)+1.6811};
    \addplot[c4000, dashed, thick, domain=0.9585:3.1987,  samples=2, forget plot] {-1.9106*log10(x)+0.9648};

    % --- D0 extrapolees ---
    \draw[c2000, very thick] (axis cs:17.3605,-0.045) -- (axis cs:17.3605,0.045);
    \draw[c3000, very thick] (axis cs:13.4513,-0.045) -- (axis cs:13.4513,0.045);
    \draw[c4000, very thick] (axis cs:3.1987,-0.045)  -- (axis cs:3.1987,0.045);

    % --- point aberrant ---

  \end{axis}
